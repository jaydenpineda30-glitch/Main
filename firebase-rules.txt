// FIREBASE FIRESTORE SECURITY RULES — Jayden's Dashboard
//
// WHERE TO PASTE THIS:
//   Firebase Console → your project → Firestore Database → Rules tab
//   Replace everything in the editor → Publish
//
// WHAT THESE DO:
//   - Only YOU (logged in) can read or write your data.
//   - Nobody else — not other users, not anonymous visitors — can read or
//     touch any document in the database.
//
// WHY NO FIELD VALIDATION HERE:
//   Data-shape checks (e.g. "an expense amount must be positive", "a task needs
//   a name") run in the app, in data-validator.js, BEFORE anything is saved.
//   They are intentionally NOT duplicated in these rules: Firestore evaluates
//   all matching rules with OR logic, so the owner-only catch-all below would
//   override any per-field rule anyway. Field rules here would be dead code that
//   lies about what it enforces, so they were removed (2026-06-24).
//
// DATA STRUCTURE (everything lives under your user namespace):
//   users/{uid}                    ← main doc (dashData field: tasks, reflections,
//                                     finance, gym, settings, boardroom, …)
//   users/{uid}/captures/{docId}   ← Quick Capture notes
//   users/{uid}/usage/{deviceId}   ← per-device interaction counts
//   users/{uid}/{anything}/{docId} ← any future subcollection (covered automatically)
// ════════════════════════════════════════════════════════════════════════════

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Is the requester logged in AND the owner of this data?
    function isOwner(uid) {
      return request.auth != null && request.auth.uid == uid;
    }

    // Deny everything by default. Anything not explicitly allowed below is blocked.
    match /{document=**} {
      allow read, write: if false;
    }

    // Your data — the user document itself, and everything beneath it.
    match /users/{uid} {
      allow read, write: if isOwner(uid);

      // Every subcollection under your user (captures, usage, and any future
      // one) — owner-only. New collections work with no rules change needed.
      match /{collection}/{docId} {
        allow read, write: if isOwner(uid);
      }
    }

  }
}
